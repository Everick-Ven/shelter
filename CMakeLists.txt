cmake_minimum_required(VERSION 3.24)
project(SHELTER VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(SHELTER_CEF_ROOT "${CMAKE_SOURCE_DIR}/cef" CACHE PATH "CEF distribution")
set(CEF_ROOT "${SHELTER_CEF_ROOT}")
set(CMAKE_MODULE_PATH "${SHELTER_CEF_ROOT}/cmake" ${CMAKE_MODULE_PATH})
find_package(CEF REQUIRED)
if(MSVC)
  set(CMAKE_CXX20_STANDARD_COMPILE_OPTION "/std:c++20")
  set(CMAKE_CXX17_STANDARD_COMPILE_OPTION "/std:c++17")
  foreach(_cfg CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE CMAKE_CXX_FLAGS_RELWITHDEBINFO)
    string(REPLACE "-std:c++20" "/std:c++20" ${_cfg} "${${_cfg}}")
  endforeach()
endif()
if(NOT EXISTS "${SHELTER_CEF_ROOT}/include/cef_app.h")
  message(FATAL_ERROR "CEF not found. Run scripts/bootstrap-cef.sh")
endif()
add_subdirectory("${SHELTER_CEF_ROOT}/libcef_dll" "${CMAKE_BINARY_DIR}/libcef_dll")
add_library(shelter_common STATIC app/common/logging.cc app/common/url_utils.cc app/browser/tab_manager.cc app/browser/browser_controller.cc app/cef/cef_app.cc app/cef/cef_client.cc app/cef/ui_scheme.cc)
target_include_directories(shelter_common PUBLIC ${CMAKE_SOURCE_DIR} ${SHELTER_CEF_ROOT})
if(WIN32)
  target_compile_definitions(shelter_common PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(WIN32)
  set(PLATFORM app/platform/windows/platform_windows.cc)
elseif(APPLE)
  set(PLATFORM app/platform/macos/platform_macos.mm)
else()
  message(FATAL_ERROR "Windows x64 and macOS Intel only")
endif()
add_executable(shelter app/main/main_common.cc ${PLATFORM})
target_link_libraries(shelter PRIVATE shelter_common libcef_dll_wrapper ${CEF_STANDARD_LIBS})
if(WIN32)
  target_link_libraries(shelter PRIVATE "${SHELTER_CEF_ROOT}/Release/libcef.lib")
endif()
target_include_directories(shelter PRIVATE ${CMAKE_SOURCE_DIR} ${SHELTER_CEF_ROOT} ${SHELTER_CEF_ROOT}/include)
if(APPLE)
  set_target_properties(shelter PROPERTIES MACOSX_BUNDLE TRUE)
  target_link_libraries(shelter PRIVATE "-framework Cocoa")
endif()
enable_testing()
add_executable(shelter_url_tests tests/url_utils_test.cc app/common/url_utils.cc)
add_executable(shelter_tab_tests tests/tab_manager_test.cc app/browser/tab_manager.cc)
target_include_directories(shelter_tab_tests PRIVATE ${CMAKE_SOURCE_DIR} ${SHELTER_CEF_ROOT})
if(WIN32)
  target_compile_definitions(shelter_tab_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_test(NAME shelter_tab_tests COMMAND shelter_tab_tests)
target_include_directories(shelter_url_tests PRIVATE ${CMAKE_SOURCE_DIR})
add_test(NAME shelter_url_tests COMMAND shelter_url_tests)
